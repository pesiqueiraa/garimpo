# Garimpo v1.2 — Veredito BR real + Kit de validação: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Veredito de concorrência BR com busca real (Serper.dev) e evidências no dossiê, kit de validação para dossiês com score final ≥ 8, e comando manual `--investigar`.

**Architecture:** Dois módulos novos de borda — `app/search.py` (Serper + LLM de veredito) e `app/kit.py` (LLM do kit) — orquestrados por `excavate._excavate_one` como etapas opcionais do dossiê. `dossiers` ganha 4 colunas (migração idempotente). `main.py` ganha o modo `--investigar` que reusa `excavate.investigate` e `report.build_dossier_page`.

**Tech Stack:** Python 3.12, SQLite, `requests` (Serper via REST, sem SDK), `google-play-scraper`, OpenAI `gpt-5-mini` via `app/llm.py`, pytest.

**Spec:** `docs/superpowers/specs/2026-07-13-v1.2-veredito-real-kit-design.md`

## Global Constraints

- Dependências de runtime: SOMENTE `requests` e `google-play-scraper` (Serper é REST puro).
- LLM: `gpt-5-mini` via `llm.ask_json`; reasoning default ("minimal") para veredito, `"medium"` para ângulo e kit.
- Serper: POST `https://google.serper.dev/search`, header `X-API-KEY`, body `{"q": ..., "gl": "br", "hl": "pt"}`, timeout 10s, top 8 resultados orgânicos por busca, 2 buscas por dossiê.
- Env vars SEMPRE lidas com `.strip()` (lição do incidente dos secrets com `\n`).
- Teto de custo por dossiê: US$0,25 — soma veredito e kit; estourou → `failed` (v1.1, inalterado).
- Veredito real e kit são etapas OPCIONAIS: falha delas nunca falha o dossiê nem o pipeline. Logs `[busca]` e `[kit]` em português.
- Kit só quando `score_final >= 8` (constante `KIT_SCORE_MIN = 8`).
- `--investigar` ignora fila/score/limite diário, salva HTML local, sem e-mail; falha → exit 1.
- Datas `datetime.now(timezone.utc).isoformat(timespec="seconds")`.
- Todo conteúdo dinâmico no HTML do e-mail passa por `html.escape`; URL de evidência só vira link se começar com `http`.
- Testes: `python3 -m pytest` a partir da raiz; mock nas bordas (`requests.post`, `llm.ask_json`, funções de `search`/`kit`), banco SQLite real temporário.
- Commits em português, rodapé `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`; commitar SOMENTE os arquivos da task.

```python
"""Busca real no Google BR (Serper.dev) para o veredito de concorrencia.

Etapa opcional da escavacao: sem chave ou com erro, o dossie segue com o
veredito-palpite do enriquecimento. Nunca derruba o pipeline.
"""
import os

import requests

from app import llm

SERPER_URL = "https://google.serper.dev/search"
TIMEOUT = 10
RESULTS_PER_QUERY = 8

VERDICT_PROMPT = """Voce e um analista de mercado brasileiro. Avalie a concorrencia
BRASILEIRA para o produto abaixo usando SOMENTE os resultados de busca reais
(Google BR) e os players de Play Store listados. Responda SOMENTE JSON:
{{"verdict": "sem_concorrente" | "concorrente_fraco" | "concorrente_forte",
 "evidence": [{{"title": "titulo do resultado", "url": "url do resultado",
                "why": "por que e evidencia, em 1 frase"}}],
 "score_final": <0-10, score recalibrado: tracao la fora + dores encontradas
                 + espaco real no Brasil>,
 "rationale": "1-2 frases justificando o veredito"}}
Regras: 2-3 itens em evidence; use SOMENTE urls presentes nos resultados
abaixo (nunca invente); se os resultados nao mostram player BR direto,
verdict="sem_concorrente" e evidence aponta os resultados que provam a busca.

Produto: {name} — {tagline}
Nicho: {niche}
Score atual (palpite, sem busca): {score}
Players na Play Store: {players_block}

Resultados da busca 1 ("{q1}"):
{results1}

Resultados da busca 2 ("{q2}"):
{results2}"""


def search_br(query: str, n: int = RESULTS_PER_QUERY) -> list[dict]:
    """Top resultados organicos do Google BR. Erros propagam para o chamador."""
    api_key = (os.environ.get("SERPER_API_KEY") or "").strip()
    r = requests.post(
        SERPER_URL,
        headers={"X-API-KEY": api_key, "Content-Type": "application/json"},
        json={"q": query, "gl": "br", "hl": "pt"},
        timeout=TIMEOUT,
    )
    r.raise_for_status()
    return [{"title": o.get("title", ""), "url": o.get("link", ""),
             "snippet": o.get("snippet", "")}
            for o in r.json().get("organic", [])[:n]]


def _results_block(results: list[dict]) -> str:
    if not results:
        return "(sem resultados)"
    return "\n".join(f"- {r['title']} — {r['url']} — {r['snippet']}" for r in results)


def verify_br_competition(product, players_block: str) -> tuple[dict, int, int] | None:
    """Veredito de concorrencia BR com busca real. None = etapa pulada."""
    if not (os.environ.get("SERPER_API_KEY") or "").strip():
        print("[busca] SERPER_API_KEY ausente, veredito continua palpite")
        return None

    q1 = f"app {product['niche'] or product['name']} Brasil"
    q2 = f"{product['pain'] or product['tagline'] or product['name']} Brasil"
    results1, results2 = [], []
    for q, dest in ((q1, results1), (q2, results2)):
        try:
            dest.extend(search_br(q))
        except Exception as e:
            print(f"[busca] serper falhou para '{q}': {e}")
    if not results1 and not results2:
        return None

    try:
        data, t_in, t_out = llm.ask_json(VERDICT_PROMPT.format(
            name=product["name"], tagline=product["tagline"] or "",
            niche=product["niche"] or "n/d",
            score=product["opportunity_score"] if product["opportunity_score"] is not None else "n/d",
            players_block=players_block,
            q1=q1, results1=_results_block(results1),
            q2=q2, results2=_results_block(results2)),
            max_tokens=1000)
    except Exception as e:
        print(f"[busca] veredito falhou: {e}")
        return None

    if data.get("verdict") not in ("sem_concorrente", "concorrente_fraco", "concorrente_forte"):
        print(f"[busca] veredito invalido do LLM: {data.get('verdict')!r}")
        return None
    return data, t_in, t_out
```

"""Kit de validacao: landing copy, perguntas Mom Test e abordagem WhatsApp.

Gerado apenas para dossies fortes (score final >= 8). Etapa opcional:
falha do kit nunca bloqueia o dossie.
"""
from app import llm

KIT_PROMPT = """Voce e um growth hacker brasileiro preparando a validacao de rua
de uma ideia de SaaS tropicalizada. Com base no dossie abaixo, gere o kit.
Responda SOMENTE JSON:
{{"landing": {{"headline": "titulo curto e forte",
   "subheadline": "1 frase de apoio",
   "benefits": ["beneficio 1", "beneficio 2", "beneficio 3"],
   "cta": "chamada para lista de espera"}},
 "mom_test": ["pergunta 1", "pergunta 2", "pergunta 3", "pergunta 4", "pergunta 5"],
 "whatsapp": "mensagem de abertura para grupo de WhatsApp do nicho"}}
Regras: tudo em portugues do Brasil, especifico do nicho (nada generico);
mom_test = exatamente 5 perguntas sobre comportamento PASSADO e real
(estilo Mom Test — nunca "voce usaria?" ou "voce pagaria?");
whatsapp = tom natural de colega do nicho, sem parecer spam, sem link.

Produto: {name} — {tagline}
Nicho: {niche} | Dor: {pain}
Mapa de dores dos concorrentes: {clusters_block}
Veredito de concorrencia BR: {verdict_block}"""


def _valid(data: dict) -> bool:
    landing = data.get("landing") or {}
    return (all(landing.get(k) for k in ("headline", "subheadline", "cta"))
            and isinstance(landing.get("benefits"), list) and len(landing["benefits"]) == 3
            and isinstance(data.get("mom_test"), list) and len(data["mom_test"]) == 5
            and bool(data.get("whatsapp")))


def build_kit(product, clusters_block: str, verdict_block: str) -> tuple[dict, int, int] | None:
    """Kit de validacao via LLM. None = kit falhou (dossie segue sem ele)."""
    try:
        data, t_in, t_out = llm.ask_json(KIT_PROMPT.format(
            name=product["name"], tagline=product["tagline"] or "",
            niche=product["niche"] or "n/d", pain=product["pain"] or "n/d",
            clusters_block=clusters_block, verdict_block=verdict_block),
            max_tokens=2000, reasoning_effort="medium")
    except Exception as e:

def investigate(conn, product_id: int) -> bool:
    """Escava um produto especifico na hora — ignora fila, score e limite diario."""

    if product is None:
        print(f"[escava] produto id={product_id} nao existe")
        return False
    try:
        _excavate_one(conn, product)

        return False

def investigate(term: str, conn=None) -> int:
    """Modo manual: escava um produto na hora e salva HTML local. 0 = sucesso."""
    conn = conn or db.connect()
    row = conn.execute(
        "SELECT * FROM products WHERE url=? OR lower(name)=lower(?)", (term, term)
    ).fetchone()
    if row is None:
        row = conn.execute(
            "SELECT * FROM products WHERE name LIKE ? ORDER BY points DESC",
            (f"%{term}%",),
        ).fetchone()
    if row is None:
        db.upsert(conn, {"name": term,
                         "url": term if term.startswith("http") else None,
                         "source": "manual", "points": 0})
        row = conn.execute("SELECT * FROM products WHERE name=?", (term,)).fetchone()
        print(f"[investigar] produto novo criado: '{term}'")
    if row["niche"] is None:
        enrich.enrich_rows(conn, only_id=row["id"])
        row = conn.execute("SELECT * FROM products WHERE id=?", (row["id"],)).fetchone()

    if not excavate.investigate(conn, row["id"]):
        return 1

    d = next(x for x in db.dossiers_since(conn, "2000-01-01T00:00:00")
             if x["dossier"]["product_id"] == row["id"])
    slug = re.sub(r"[^a-z0-9]+", "-", row["name"].lower()).strip("-")[:40] or "produto"
    out = db.DB_PATH.parent / f"dossie-{slug}-{date.today().isoformat()}.html"
    out.parent.mkdir(exist_ok=True)
    out.write_text(report.build_dossier_page(d), encoding="utf-8")
    print(f"[investigar] dossie salvo em {out}")
