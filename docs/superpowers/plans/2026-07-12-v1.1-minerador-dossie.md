# Garimpo v1.1 — Minerador de Reviews + Dossiê: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Escavação automática de concorrentes na Play Store para produtos com score ≥ 7, clusterização de dores via LLM e dossiê completo no e-mail diário.

**Architecture:** A escavação é o passo 3 do pipeline diário existente (`main.py`), com estado em colunas novas de `products` e três tabelas novas. Módulo novo `app/llm.py` unifica todas as chamadas OpenAI; `app/excavate.py` orquestra fila → sugestão → Play Store → reviews → clusters → ângulo de ataque.

**Tech Stack:** Python 3.12, SQLite, `requests`, `google-play-scraper` (novo), OpenAI `gpt-5-mini`, pytest (novo, dev-only).

**Spec:** `docs/superpowers/specs/2026-07-12-v1.1-minerador-dossie-design.md`

## Global Constraints

- Dependências de runtime: SOMENTE `requests` e `google-play-scraper`. pytest fica em `requirements-dev.txt`.
- Modelo LLM: `gpt-5-mini`; `reasoning_effort: "minimal"` para sugestão/clusterização, `"medium"` para o ângulo de ataque; sempre `response_format: {"type": "json_object"}`.
- Máx. 2 dossiês por run (`MAX_DOSSIERS_PER_RUN = 2`); fila = `opportunity_score >= 7 AND dig_status IS NULL AND dig_attempts < 3`.
- Play Store: loja BR (`country="br", lang="pt"`) primeiro, US (`country="us", lang="en"`) como fallback; ~70 reviews por estrela (1★, 2★, 3★); pausa de 1.5s entre requisições.
- Teto de custo por dossiê: US$0,25 — estourou, aborta e marca `failed` direto.
- Falha graciosa em 3 níveis: concorrente → dossiê → etapa. A escavação NUNCA derruba o pipeline.
- Logs em português com prefixo `[escava]`, seguindo o padrão `[fonte] mensagem` dos módulos existentes.
- Reviews cruas NÃO são persistidas no banco.
- Datas sempre `datetime.now(timezone.utc).isoformat(timespec="seconds")`, como no código atual.
- Rodar testes sempre com `python3 -m pytest` a partir da raiz (garante `app` no sys.path).
- Commits frequentes, mensagens em português, rodapé `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.
- **Antes de qualquer trabalho: `git pull`** (o workflow diário commita `data/garimpo.db`).

```python
def _migrate(conn: sqlite3.Connection) -> None:
    """Migracao leve: bancos da v1.0 ganham as colunas de escavacao."""
    cols = {r[1] for r in conn.execute("PRAGMA table_info(products)")}
    if "dig_status" not in cols:
        conn.execute("ALTER TABLE products ADD COLUMN dig_status TEXT")
    if "dig_attempts" not in cols:
        conn.execute("ALTER TABLE products ADD COLUMN dig_attempts INTEGER DEFAULT 0")
    conn.commit()


def dig_queue(conn: sqlite3.Connection, limit: int = 2) -> list[sqlite3.Row]:
    """Produtos elegiveis para escavacao, melhores primeiro."""
    return conn.execute(
        """SELECT * FROM products
           WHERE opportunity_score >= 7 AND dig_status IS NULL AND dig_attempts < 3
           ORDER BY opportunity_score DESC, points DESC LIMIT ?""",
        (limit,),
    ).fetchall()


def mark_dig(conn: sqlite3.Connection, product_id: int, status: str) -> None:
    conn.execute("UPDATE products SET dig_status=? WHERE id=?", (status, product_id))
    conn.commit()


def record_dig_failure(conn: sqlite3.Connection, product_id: int) -> None:
    conn.execute(
        """UPDATE products SET dig_attempts = dig_attempts + 1,
           dig_status = CASE WHEN dig_attempts + 1 >= 3 THEN 'failed' ELSE dig_status END
           WHERE id=?""",
        (product_id,),
    )
    conn.commit()


def _now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def save_competitors(conn: sqlite3.Connection, product_id: int, comps: list[dict]) -> None:
    now = _now()
    conn.execute("DELETE FROM competitors WHERE product_id=?", (product_id,))
    conn.executemany(
        """INSERT INTO competitors
           (product_id, app_id, name, store, rating, ratings_count, reviews_collected, created_at)
           VALUES (?,?,?,?,?,?,?,?)""",
        [(product_id, c["app_id"], c["name"], c["store"], c.get("rating"),
          c.get("ratings_count"), c.get("reviews_collected", 0), now) for c in comps],
    )
    conn.commit()


def save_pain_clusters(conn: sqlite3.Connection, product_id: int, clusters: list[dict]) -> None:
    now = _now()
    conn.execute("DELETE FROM pain_clusters WHERE product_id=?", (product_id,))
    conn.executemany(
        """INSERT INTO pain_clusters
           (product_id, theme, frequency, severity, examples, created_at)
           VALUES (?,?,?,?,?,?)""",
        [(product_id, c["theme"], c.get("frequency", 0), c.get("severity"),
          json.dumps(c.get("examples", []), ensure_ascii=False), now) for c in clusters],
    )
    conn.commit()


def save_dossier(conn: sqlite3.Connection, product_id: int, attack_angle: str,
                 cost_usd: float, tokens_in: int, tokens_out: int) -> None:
    conn.execute(
        """INSERT OR REPLACE INTO dossiers
           (product_id, attack_angle, cost_usd, tokens_in, tokens_out, created_at)
           VALUES (?,?,?,?,?,?)""",
        (product_id, attack_angle, cost_usd, tokens_in, tokens_out, _now()),
    )
    conn.commit()


def dossiers_since(conn: sqlite3.Connection, iso_cutoff: str) -> list[dict]:
    """Dossies recentes com produto, players e clusters — prontos para o relatorio."""
    rows = conn.execute(
        """SELECT d.*, p.name, p.tagline, p.url, p.source_url, p.source, p.sources_seen,
                  p.niche, p.pain, p.opportunity_score, p.points
           FROM dossiers d JOIN products p ON p.id = d.product_id
           WHERE d.created_at >= ? ORDER BY d.created_at DESC""",
        (iso_cutoff,),
    ).fetchall()
    return [{
        "dossier": r,
        "competitors": conn.execute(
            "SELECT * FROM competitors WHERE product_id=? ORDER BY ratings_count DESC",
            (r["product_id"],)).fetchall(),
        "clusters": conn.execute(
            "SELECT * FROM pain_clusters WHERE product_id=? ORDER BY frequency DESC",
            (r["product_id"],)).fetchall(),
    } for r in rows]
```

```python
    path = db_path or DB_PATH
    path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(path)
```

```python
"""Porta unica para a API da OpenAI (Chat Completions).

Toda chamada LLM do projeto passa por ask_json: JSON garantido,
retry de resposta malformada e contagem de tokens para medir custo.
"""
import json
import os

import requests

MODEL = "gpt-5-mini"
API_URL = "https://api.openai.com/v1/chat/completions"
TIMEOUT = 90
# USD por 1M tokens (gpt-5-mini) — para estimativa de custo, nao cobranca
PRICE_IN_PER_M = 0.25
PRICE_OUT_PER_M = 2.00


class LLMError(Exception):
    """Chave ausente ou resposta invalida apos retry."""


def cost_usd(tokens_in: int, tokens_out: int) -> float:
    return tokens_in / 1e6 * PRICE_IN_PER_M + tokens_out / 1e6 * PRICE_OUT_PER_M


def ask_json(prompt: str, max_tokens: int = 500,
             reasoning_effort: str = "minimal") -> tuple[dict, int, int]:
    """Pergunta ao modelo e retorna (dados, tokens_in, tokens_out).

    1 retry automatico se a resposta nao for JSON valido; tokens dos
    dois attempts sao somados. Erros HTTP propagam (requests.RequestException).
    """
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        raise LLMError("OPENAI_API_KEY ausente")

    messages = [{"role": "user", "content": prompt}]
    total_in = total_out = 0
    for attempt in (1, 2):
        r = requests.post(
            API_URL,
            headers={"Authorization": f"Bearer {api_key}",
                     "content-type": "application/json"},
            json={
                "model": MODEL,
                "reasoning_effort": reasoning_effort,
                "response_format": {"type": "json_object"},
                "max_completion_tokens": max_tokens,
                "messages": messages,
            },
            timeout=TIMEOUT,
        )
        r.raise_for_status()
        payload = r.json()
        text = payload["choices"][0]["message"]["content"]
        usage = payload.get("usage", {})
        total_in += usage.get("prompt_tokens", 0)
        total_out += usage.get("completion_tokens", 0)
        try:
            data = json.loads(text.replace("```json", "").replace("```", "").strip())
            return data, total_in, total_out
        except json.JSONDecodeError:
            if attempt == 2:
                raise LLMError(f"JSON invalido apos retry: {text[:200]!r}")
            messages.append({"role": "assistant", "content": text})
            messages.append({"role": "user",
                             "content": "A resposta anterior nao era JSON valido. "
                                        "Responda SOMENTE o objeto JSON, sem texto extra."})
```

```python
Classifica nicho, dor, publico e da um palpite inicial sobre concorrencia
no Brasil (baseado no conhecimento do modelo — para veredito confiavel,
o proximo passo e plugar uma API de busca como Serper/Tavily).

Requer OPENAI_API_KEY no ambiente. Sem a chave, o passo e pulado.
"""
import os

from app import llm

PROMPT = """Voce e um analista de oportunidades de SaaS para o mercado brasileiro.
Analise o produto abaixo e responda SOMENTE com JSON valido, sem markdown:

{{"niche": "nicho em 2-4 palavras em portugues",
 "pain": "dor que resolve, em uma frase curta em portugues",
 "audience": "B2B" ou "B2C" ou "ambos",
 "br_verdict": "sem_concorrente" | "concorrente_fraco" | "concorrente_forte" | "desconhecido",
 "opportunity_score": numero de 0 a 10}}

Criterios do score: tracao aparente + ausencia de player brasileiro conhecido
+ viabilidade de adaptar ao Brasil (Pix, WhatsApp, NF etc). Se nao tiver
certeza sobre concorrentes BR, use "desconhecido" e seja conservador.

Produto: {name}
Descricao: {tagline}
Categoria: {category}
Fonte: {source} ({points} pontos/votos)"""


def enrich_rows(conn, limit: int = 25) -> int:
    if not os.environ.get("OPENAI_API_KEY"):
        print("[enrich] OPENAI_API_KEY ausente, pulando enriquecimento")
        return 0

    rows = conn.execute(
        "SELECT * FROM products WHERE niche IS NULL ORDER BY points DESC LIMIT ?",
        (limit,),
    ).fetchall()

    done = 0
    for row in rows:
        try:
            data, _, _ = llm.ask_json(PROMPT.format(
                name=row["name"], tagline=row["tagline"] or "",
                category=row["category"] or "n/d",
                source=row["source"], points=row["points"],
            ))
            conn.execute(
                """UPDATE products SET niche=?, pain=?, audience=?,
                   br_verdict=?, opportunity_score=? WHERE id=?""",
                (data.get("niche"), data.get("pain"), data.get("audience"),
                 data.get("br_verdict"), data.get("opportunity_score"), row["id"]),
            )
            conn.commit()
            done += 1
        except Exception as e:
            print(f"[enrich] falhou para '{row['name']}': {e}")
    print(f"[enrich] {done} produtos enriquecidos")
    return done
```

```python
"""Escavacao de concorrentes: Play Store + LLM viram dossie.

Fluxo por produto: sugestao de concorrentes (LLM) -> confirmacao na
Play Store (BR primeiro, US fallback) -> reviews 1-3 estrelas ->
clusterizacao de dores (LLM) -> angulo de ataque (LLM).
Reviews cruas nao sao persistidas; so o destilado vai para o banco.
"""
import time

from google_play_scraper import app as gp_app
from google_play_scraper import reviews as gp_reviews
from google_play_scraper import search as gp_search
from google_play_scraper import Sort

from app import db, llm

MAX_DOSSIERS_PER_RUN = 2
REVIEWS_PER_STAR = 70
REVIEW_CHAR_LIMIT = 300
CLUSTER_REVIEW_CAP = 150
PAUSE_S = 1.5
COST_CEILING_USD = 0.25

STORE_LANG = {"br": "pt", "us": "en"}


class BudgetExceeded(Exception):
    """Dossie passou do teto de custo em tokens."""


def confirm_on_play_store(search_term: str) -> dict | None:
    """Busca o termo na loja BR; sem resultado, tenta US. None se nada."""
    for store in ("br", "us"):
        lang = STORE_LANG[store]
        try:
            hits = gp_search(search_term, lang=lang, country=store, n_hits=3)
        except Exception as e:
            print(f"[escava] busca '{search_term}' ({store}) falhou: {e}")
            continue
        if not hits:
            continue
        hit = hits[0]
        details = {}
        try:
            details = gp_app(hit["appId"], lang=lang, country=store)
        except Exception:
            pass  # segue com os dados da busca
        time.sleep(PAUSE_S)
        return {
            "app_id": hit["appId"],
            "name": hit.get("title", search_term),
            "store": store,
            "rating": details.get("score", hit.get("score")),
            "ratings_count": details.get("ratings"),
        }
    return None


def collect_negative_reviews(app_id: str, store: str) -> list[str]:
    """Ate ~70 reviews por estrela (1, 2, 3), com pausa entre requisicoes."""
    lang = STORE_LANG[store]
    texts = []
    for star in (1, 2, 3):
        try:
            batch, _ = gp_reviews(app_id, lang=lang, country=store,
                                  count=REVIEWS_PER_STAR,
                                  filter_score_with=star, sort=Sort.NEWEST)
        except Exception as e:
            print(f"[escava] reviews {star}★ de {app_id} falharam: {e}")
            batch = []
        texts += [r["content"] for r in batch if r.get("content")]
        time.sleep(PAUSE_S)
    return texts


def sample_reviews(reviews_by_app: dict[str, list[str]],
                   cap: int = CLUSTER_REVIEW_CAP) -> dict[str, list[str]]:
    """Trunca cada review e limita o total, dividido igualmente entre apps."""
    if not reviews_by_app:
        return {}
    per_app = max(1, cap // len(reviews_by_app))
    return {name: [r[:REVIEW_CHAR_LIMIT] for r in revs[:per_app]]
            for name, revs in reviews_by_app.items()}
```

```python
SUGGEST_PROMPT = """Voce e um analista de mercado de SaaS. Para o produto abaixo,
liste de 3 a 5 aplicativos concorrentes ou similares que provavelmente existem
na Google Play Store — inclua players globais E brasileiros, se conhecer.
Responda SOMENTE JSON:
{{"competitors": [{{"name": "nome do app", "search_term": "termo curto de busca"}}]}}

Produto: {name}
Descricao: {tagline}
Nicho: {niche}
Dor que resolve: {pain}"""

CLUSTER_PROMPT = """Voce e um analista de produto. Abaixo estao reviews negativas
(1-3 estrelas) de apps concorrentes no nicho "{niche}". Agrupe as reclamacoes
em 3 a 7 temas de dor, em portugues. Responda SOMENTE JSON:
{{"clusters": [{{"theme": "nome curto do tema",
  "frequency": <n de reviews que citam o tema>,
  "severity": "alta" | "media" | "baixa",
  "examples": [{{"app": "nome do app", "excerpt": "trecho literal curto"}}]}}]}}
Ordene por frequency decrescente. 2-3 examples por tema.

Reviews por app:
{reviews_block}"""

ANGLE_PROMPT = """Voce e um estrategista de SaaS focado em tropicalizar produtos
validados la fora para o Brasil (Pix, WhatsApp, nota fiscal, Simples Nacional).

Produto original: {name} — {tagline} (nicho: {niche})
Players encontrados na Play Store: {players_block}
Mapa de dores extraido de reviews 1-3 estrelas: {clusters_block}

Escreva o "angulo de ataque" para um empreendedor brasileiro: quais dores
negligenciadas atacar e como adaptar ao Brasil. Um paragrafo denso (max 120
palavras) seguido de 3 bullets de diferenciais concretos, separados por \\n.
Responda SOMENTE JSON: {{"attack_angle": "texto"}}"""


def _reviews_block(sampled: dict[str, list[str]]) -> str:
    parts = []
    for app_name, revs in sampled.items():
        joined = "\n".join(f"- {r}" for r in revs)
        parts.append(f"## {app_name}\n{joined}")
    return "\n\n".join(parts)


def _check_budget(tokens: dict) -> None:
    spent = llm.cost_usd(tokens["in"], tokens["out"])
    if spent > COST_CEILING_USD:
        raise BudgetExceeded(f"US${spent:.2f} > teto US${COST_CEILING_USD}")


def _excavate_one(conn, product) -> None:
    tokens = {"in": 0, "out": 0}

    data, t_in, t_out = llm.ask_json(SUGGEST_PROMPT.format(
        name=product["name"], tagline=product["tagline"] or "",
        niche=product["niche"] or "n/d", pain=product["pain"] or "n/d"))
    tokens["in"] += t_in; tokens["out"] += t_out
    _check_budget(tokens)
    suggestions = data.get("competitors", [])[:5]

    competitors, reviews_by_app, seen = [], {}, set()
    for s in suggestions:
        comp = confirm_on_play_store(s.get("search_term") or s.get("name", ""))
        if not comp or comp["app_id"] in seen:
            continue
        seen.add(comp["app_id"])
        revs = collect_negative_reviews(comp["app_id"], comp["store"])
        comp["reviews_collected"] = len(revs)
        competitors.append(comp)
        if revs:
            reviews_by_app[comp["name"]] = revs

    clusters = []
    if reviews_by_app:
        data, t_in, t_out = llm.ask_json(
            CLUSTER_PROMPT.format(niche=product["niche"] or "n/d",
                                  reviews_block=_reviews_block(sample_reviews(reviews_by_app))),
            max_tokens=2000)
        tokens["in"] += t_in; tokens["out"] += t_out
        _check_budget(tokens)
        clusters = data.get("clusters", [])

    players_block = ("; ".join(
        f"{c['name']} ({'BR' if c['store'] == 'br' else 'global'}, "
        f"nota {c['rating']}, {c['ratings_count'] or '?'} avaliacoes)"
        for c in competitors) or "nenhum app encontrado na Play Store")
    clusters_block = ("; ".join(
        f"{c['theme']} ({c['frequency']}x, {c['severity']})" for c in clusters)
        or "sem dados de reviews")

    data, t_in, t_out = llm.ask_json(
        ANGLE_PROMPT.format(name=product["name"], tagline=product["tagline"] or "",
                            niche=product["niche"] or "n/d",
                            players_block=players_block, clusters_block=clusters_block),
        max_tokens=1500, reasoning_effort="medium")
    tokens["in"] += t_in; tokens["out"] += t_out
    _check_budget(tokens)

    cost = llm.cost_usd(tokens["in"], tokens["out"])
    db.save_competitors(conn, product["id"], competitors)
    db.save_pain_clusters(conn, product["id"], clusters)
    db.save_dossier(conn, product["id"], data.get("attack_angle", ""),
                    cost, tokens["in"], tokens["out"])
    db.mark_dig(conn, product["id"], "done")
    print(f"[escava] dossie de '{product['name']}': {len(competitors)} players, "
          f"{len(clusters)} dores, US${cost:.4f}")


def excavate_all(conn) -> int:
    """Escava a fila do dia. Nunca levanta excecao — falha graciosa."""
    if not os.environ.get("OPENAI_API_KEY"):
        print("[escava] OPENAI_API_KEY ausente, pulando escavacao")
        return 0
    queue = db.dig_queue(conn, MAX_DOSSIERS_PER_RUN)
    if not queue:
        print("[escava] fila vazia (nenhum produto com score >= 7 pendente)")
        return 0

    done = 0
    for product in queue:
        try:
            _excavate_one(conn, product)
            done += 1
        except BudgetExceeded as e:
            print(f"[escava] '{product['name']}' estourou orcamento: {e}")
            db.mark_dig(conn, product["id"], "failed")
        except Exception as e:
            print(f"[escava] falhou para '{product['name']}': {e}")
            db.record_dig_failure(conn, product["id"])
    return done
```
